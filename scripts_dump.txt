        function saveSingleOwner(btn) {
            const row = btn.closest('.owner-item');
            const projectId = document.querySelector('input[name="DraftProjectId"]').value;
            
            // Collect data
            const firstName = row.querySelector('input[name$=".FirstName"]').value;
            const lastName = row.querySelector('input[name$=".LastName"]').value;
            const phone = row.querySelector('input[name$=".PhoneNumber"]').value;
            const email = row.querySelector('input[name$=".Email"]').value;
            const blockName = row.querySelector('select[name$=".BlockName"]').value;
            const unitType = row.querySelector('select[name$=".UnitType"]').value;
            const flatNumber = row.querySelector('input[name$=".FlatNumber"]').value;
            const isRep = row.querySelector('input[name$=".IsRepresentative"]').checked;

            if(!firstName || !lastName) {
                alert("Lütfen en azından Ad ve Soyad giriniz.");
                return;
            }

            const formData = new FormData();
            // Get anti-forgery token
            const token = document.querySelector('input[name="__RequestVerificationToken"]').value;
            formData.append('__RequestVerificationToken', token);
            formData.append('projectId', projectId);
            formData.append('firstName', firstName);
            formData.append('lastName', lastName);
            formData.append('phone', phone);
            formData.append('email', email);
            formData.append('blockName', blockName);
            formData.append('unitType', unitType);
            formData.append('flatNumber', flatNumber);
            formData.append('isRep', isRep);

            // Visual feedback
            const originalHtml = btn.innerHTML;
            btn.innerHTML = '<span class="spinner-border spinner-border-sm" role="status" aria-hidden="true"></span>';
            btn.disabled = true;

            fetch('/ConstructionProject/SaveSingleOwner', {
                method: 'POST',
                body: formData
            })
            .then(res => res.json())
            .then(data => {
                if(data.success) {
                    btn.classList.replace('btn-outline-success', 'btn-success');
                    btn.innerHTML = '<i class="bi bi-check-all"></i>';
                    alert(firstName + " " + lastName + " başarıyla kaydedildi!");
                    setTimeout(() => {
                        btn.disabled = false;
                        btn.classList.replace('btn-success', 'btn-outline-success');
                        btn.innerHTML = '<i class="bi bi-check-lg"></i>';
                    }, 2000);
                } else {
                    alert("Kayıt hatası: " + data.message);
                    btn.disabled = false;
                    btn.innerHTML = originalHtml;
                }
            })
            .catch(err => {
                alert("Bağlantı hatası: " + err.message);
                btn.disabled = false;
                btn.innerHTML = originalHtml;
            });
        }

        function openMapModal() {
            var mapModal = new bootstrap.Modal(document.getElementById('mapModal'));
            mapModal.show();
            
            setTimeout(() => {
                if(!map) {
                    map = L.map('projectMap').setView([41.0082, 28.9784], 13);
                    L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', {
                        attribution: '© OpenStreetMap contributors'
                    }).addTo(map);
                    
                    marker = L.marker([41.0082, 28.9784], {draggable: true}).addTo(map);
                    
                    map.on('click', function(e) {
                        marker.setLatLng(e.latlng);
                    });
                } else {
                    map.invalidateSize();
                }

                var cityName = $('#CityId option:selected').text();
                var distName = $('#DistrictId option:selected').text();
                var neighName = $('#NeighborhoodId option:selected').text();
                var streetName = $('#StreetId option:selected').text();
                
                if (cityName && cityName !== 'Seçiniz') {
                    var query = '';
                    if (neighName && neighName !== 'Seçiniz' && neighName !== 'Yükleniyor...') {
                        if (streetName && streetName !== 'Seçiniz' && streetName !== 'Yükleniyor...') {
                            query = `${streetName}, ${neighName}, ${distName}, ${cityName}, Turkey`;
                        } else {
                            query = `${neighName}, ${distName}, ${cityName}, Turkey`;
                        }
                    } else if (distName && distName !== 'Seçiniz') {
                        query = `${distName}, ${cityName}, Turkey`;
                    } else {
                        query = `${cityName}, Turkey`;
                    }

                    var query = '';
                    if (neighName && neighName !== 'Seçiniz' && neighName !== 'Yükleniyor...') {
                        if (streetName && streetName !== 'Seçiniz' && streetName !== 'Yükleniyor...') {
                            query = `${streetName}, ${neighName}, ${distName}, ${cityName}, Turkey`;
                        } else {
                            query = `${neighName}, ${distName}, ${cityName}, Turkey`;
                        }
                    } else if (distName && distName !== 'Seçiniz') {
                        query = `${distName}, ${cityName}, Turkey`;
                    } else {
                        query = `${cityName}, Turkey`;
                    }

                    var tryGeocode = function(searchQuery, zoomLvl) {
                        fetch(`https://nominatim.openstreetmap.org/search?format=json&q=${encodeURIComponent(searchQuery)}&limit=1`)
                        .then(res => res.json())
                        .then(data => {
                            if(data && data.length > 0) {
                                var lat = parseFloat(data[0].lat);
                                var lon = parseFloat(data[0].lon);
                                map.setView([lat, lon], zoomLvl);
                                marker.setLatLng([lat, lon]);
                            } else {
                                // Bulamazsa bir üste geç
                                if (searchQuery.includes(streetName) && streetName) {
                                    tryGeocode(`${neighName}, ${distName}, ${cityName}, Turkey`, 15);
                                } else if (searchQuery.includes(neighName) && neighName) {
                                    tryGeocode(`${distName}, ${cityName}, Turkey`, 13);
                                }
                            }
                        });
                    };
                    
                    tryGeocode(query, 16);
                }
            }, 500);
        }

        document.getElementById('fullAddress').addEventListener('input', function(e) {
            let val = e.target.value;
            // 1. Türkçe karakter ve noktalama kalsın, garip sembolleri uçur.
            val = val.replace(/[^a-zA-Z0-9çÇğĞıİöÖşŞüÜ\s\.,:\/-]/g, '');
            
            // 2. 16a veya 16A yazılırsa 16/A'ya çevir
            val = val.replace(/(\d+)([a-zA-ZçÇğĞıİöÖşŞüÜ])/g, '$1/$2');
            
            // 3. / yanındaki harfi BÜYÜT
            val = val.replace(/\/([a-zçğıöşü])/g, function(match, p1) {
                return '/' + p1.toLocaleUpperCase('tr-TR');
            });
            
            if(e.target.value !== val) {
                const start = e.target.selectionStart;
                const end = e.target.selectionEnd;
                e.target.value = val;
                e.target.setSelectionRange(start, end);
            }
        });

                $('#StreetId').empty().append('<option value="">Yükleniyor...</option>');
                
                if (neighborhoodId) {
                    $.ajax({
                        url: '/api/Location/Streets/' + neighborhoodId,
                        method: 'GET',
                        dataType: 'json',
                        success: function(data) {
                            $('#StreetId').empty().append('<option value="">Seçiniz</option>');
                            if(data && data.length > 0) {
                                data.forEach(function(d) {
                                    var optId = d.id !== undefined ? d.id : d.Id;
                                    var optName = d.name !== undefined ? d.name : d.Name;
                                    $('#StreetId').append('<option value="' + optId + '">' + optName + '</option>');
                                });
                            }
                        },
                        error: function(xhr, status, error) {
                            alert("Sistem Uyarısı: Sokak çağrısı çöktü!\nHata Kodu: " + xhr.status + "\nMesaj: " + error);
                        }
                    });
                } else {
                    $('#StreetId').empty().append('<option value="">Seçiniz</option>');
                }
            });
        });
        function toggleParent(sel) {
              var val = $(sel).val();
              var parentSel = $(sel).siblings('.b-parent');
              if(val === 'tower') {
                  parentSel.show();
                  parentSel.attr('required', true);
              } else {
                  parentSel.hide();
                  parentSel.removeAttr('required');
                  parentSel.val('');
              }
              updatePodiumDropdowns();
          }

          function updatePodiumDropdowns() {
              var podiums = [];
              $('.block-item').each(function(index) {
                  var type = $(this).find('.b-type').val();
                  var name = $(this).find('.b-name').val();
                  if(type === 'podium' && name.trim() !== '') {
                      podiums.push({ idx: index, name: name });
                  }
              });
              
              $('.b-parent').each(function() {
                  var currentVal = $(this).val();
                  $(this).empty();
                  $(this).append('<option value="">Hangi Bazaya Bağlı?</option>');
                  var sel = $(this);
                  podiums.forEach(function(p) {
                      sel.append('<option value="' + p.idx + '">' + p.name + '</option>');
                  });
                  $(this).val(currentVal); 
              });
          }




          function prepareSummary() {
            // Get values from Step 1
            const name = document.querySelector('input[name="Name"]').value;
            const desc = document.querySelector('textarea[name="Description"]').value;
            
            // Get values from Step 2
            const blocks = document.querySelectorAll('.block-item');
            let totalApts = 0;
            let totalShops = 0;
            
            blocks.forEach(b => {
                let aptsInput = b.querySelector('input[name*="TotalApartments"]');
                let shopsInput = b.querySelector('input[name*="TotalShops"]');
                if (aptsInput) totalApts += (parseInt(aptsInput.value) || 0);
                if (shopsInput) totalShops += (parseInt(shopsInput.value) || 0);
            });


            // Set Summary
            document.getElementById('summaryName').innerText = name || "İsimsiz Proje";
            document.getElementById('summaryDesc').innerText = desc;
            document.getElementById('summaryBlockCount').innerText = blocks.length;
            document.getElementById('summaryTotalApts').innerText = totalApts;
            document.getElementById('summaryTotalShops').innerText = totalShops;
            
            nextStep(4);
        }
        
    

function handleStatusChange() {
    var statusSel = document.getElementById('StatusId');
    var helpText = document.getElementById('statusHelpText');
    if (!statusSel || !helpText) return;
    
    var val = statusSel.value;
    if (val === '2') {
        helpText.innerText = 'Aktif olarak inşaatı süren, puantaj ve maliyet takibi yapacağınız projedir.';
    } else if (val === '0') {
        helpText.innerText = 'Fizibilite, teklif veya kentsel dönüşüm sürecindeki aday projelerdir.';
    } else if (val === '3') {
        helpText.innerText = 'Tamamlanmış referans proje veya sadece bina yönetimi (aidat) yapılacak mülktür.';
    }
}
